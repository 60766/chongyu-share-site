func sharePostAsScreenshot(post: UserPostModel) {
    // 1. 截图当前帖子视图区域
    let postView = PostCardView(post: post, displayMode: .fullContent)
    
    // 2. 添加品牌水印
    let enhancedView = PostScreenshotCard(
        postView: postView,
        showBranding: true,
        showQRCode: true

#### 步骤1：创建帖子分享卡片组件
```swift
struct PostShareCard: View {
    let post: UserPostModel
    let includeQRCode: Bool
    let includeBranding: Bool
    
    var body: some View {
        VStack(alignment: .leading, spacing: 16) {
            // 品牌头部
            if includeBranding {
                brandingHeader
            }
            
            // 帖子内容（原样式）
            PostCardView(post: post, displayMode: .shareOptimized)
                .background(Color(.systemBackground))
                .cornerRadius(12)
                .shadow(color: .black.opacity(0.1), radius: 4)
            
            // 底部信息
            if includeQRCode {
                downloadPrompt
            }
        }
        .padding()
        .background(backgroundGradient)
    }
    
    private var brandingHeader: some View {
        HStack {
            Image("app_icon")
                .resizable()
                .frame(width: 28, height: 28)
                .cornerRadius(6)
            
            VStack(alignment: .leading, spacing: 2) {
                Text("虫遇")
                    .font(.headline)
                    .fontWeight(.semibold)
                Text("穿越时空对话")
                    .font(.caption)
                    .foregroundColor(.secondary)
            }
            
            Spacer()
        }
    }
    
    private var downloadPrompt: some View {
        HStack {
            VStack(alignment: .leading, spacing: 4) {
                Text("扫码下载虫遇App")
                    .font(.subheadline)
                    .fontWeight(.medium)
                Text("与历史人物对话")
                    .font(.caption)
                    .foregroundColor(.secondary)
            }
            
            Spacer()
            
            QRCodeView(
                url: "https://apps.apple.com/app/虫遇",
                size: 60
            )
            .cornerRadius(8)
        }
        .padding(.horizontal, 16)
        .padding(.vertical, 12)
        .background(Color(.systemGray6))
        .cornerRadius(12)
    }
    
    private var backgroundGradient: some View {
        LinearGradient(
            colors: [
                Color(.systemBackground),
                Color(.systemGray6).opacity(0.3)
            ],
            startPoint: .top,
            endPoint: .bottom
        )
    }
}
```

#### 步骤3：集成到现有分享流程
```swift
extension ShareManager {
    func sharePost(_ post: UserPostModel, from viewController: UIViewController) {
        // 生成分享图片
        let shareImage = generatePostShareImage(post)
        
        // 分享文本
        let shareText = generatePostShareText(post)
        
        // 调用系统分享
        let items: [Any] = [shareText, shareImage].compactMap { $0 }
        let activityVC = UIActivityViewController(activityItems: items, applicationActivities: nil)
        viewController.present(activityVC, animated: true)
    }
    
    private func generatePostShareImage(_ post: UserPostModel) -> UIImage? {
        let shareCard = PostShareCard(
            post: post,
            includeQRCode: true,
            includeBranding: true
        )
        
        return renderViewAsImage(shareCard, size: CGSize(width: 375, height: 650))
    }
    
    private func generatePostShareText(_ post: UserPostModel) -> String {
        return """
        【\(post.username)的虫遇动态】
        
        \(post.content.prefix(100))...
        
        来自虫遇App - 穿越时空的对话
        """
    }
}
```
